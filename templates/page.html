{#- Related posts — the three sharing the most tags with this one. Internal
    linking was prev/next only, which is chronology, not relevance.

    Tera can't sort by a computed score, so rather than score-then-sort this
    sweeps the candidate list once per shared-tag count, highest first, and
    stops at three. O(tags x posts) with both numbers tiny, and the resulting
    order is obvious by construction. #}
{%- if page.taxonomies and page.taxonomies.tags %}
{%- set blog = get_section(path="_index.md") %}
{%- set_global related = [] %}
{%- set_global related_urls = [] %}
{%- for needed in [5, 4, 3, 2, 1] %}
  {%- for candidate in blog.pages %}
    {%- if related | length < 3 and candidate.permalink != page.permalink and candidate.permalink not in related_urls %}
      {#- Tera won't take a filter on the right of `in`, so bind it first. #}
      {%- set candidate_tags = candidate.taxonomies.tags | default(value=[]) %}
      {%- set_global shared = 0 %}
      {%- for tag in page.taxonomies.tags %}
        {%- if tag in candidate_tags %}
          {%- set_global shared = shared + 1 %}
        {%- endif %}
      {%- endfor %}
      {%- if shared >= needed %}
        {%- set_global related = [...related, candidate] %}
        {%- set_global related_urls = [...related_urls, candidate.permalink] %}
      {%- endif %}
    {%- endif %}
  {%- endfor %}
{%- endfor %}
